cmake_minimum_required(VERSION 3.20)
if (CMAKE_HOST_APPLE AND NOT CMAKE_SYSTEM_NAME AND NOT CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
endif ()

project(huxerui_preview VERSION 0.1.0 LANGUAGES NONE)
include(HuxerUIProject.cmake)
if (HUXERUI_PROJECT_PLAN_ONLY)
    return()
endif ()

include(project-path.cmake)
include(preview-sources.cmake)
include("${HUXERUI_PREVIEW_PROJECT_DIR}/preview/preview.cmake" OPTIONAL)

set(host_sources src/app.cpp)
if (NOT HUXERUI_LIBRARY_GRAPH_ONLY)
    if (WIN32)
        list(APPEND host_sources platform/windows/main.cpp capture-windows.cpp)
    elseif (APPLE)
        enable_language(OBJCXX)
        list(APPEND host_sources platform/macos/main.cpp capture-macos.mm)
    elseif (CMAKE_SYSTEM_NAME STREQUAL "Linux")
        list(APPEND host_sources platform/linux/main.cpp capture-linux.cpp)
    endif ()
endif ()

huxerui_add_app(huxerui_preview
    SOURCES ${host_sources} ${HUXERUI_PREVIEW_SOURCES}
    BUNDLE_NAME "HuxerUI Preview"
    BUNDLE_IDENTIFIER "${HUXERUI_PREVIEW_APP_ID}"
)

if (NOT HUXERUI_LIBRARY_GRAPH_ONLY)
    foreach (prefix IN ITEMS HUXERUI_PREVIEW_RESOURCE HUXERUI_PREVIEW_LIBRARY_RESOURCE)
        if (${prefix}_NAMESPACE AND IS_DIRECTORY "${${prefix}_DIR}")
            huxerui_add_resources(huxerui_preview ROOT "${${prefix}_DIR}" NAMESPACE "${${prefix}_NAMESPACE}")
        endif ()
    endforeach ()
endif ()

if (COMMAND huxerui_configure_preview)
    huxerui_configure_preview(huxerui_preview)
endif ()
huxerui_configure_project_app(huxerui_preview)

if (HUXERUI_LIBRARY_GRAPH_ONLY)
    return()
endif ()
target_include_directories(huxerui_preview PRIVATE ${HUXERUI_PREVIEW_INCLUDE_DIRS})
set_target_properties(huxerui_preview PROPERTIES CXX_EXTENSIONS OFF)
if (APPLE)
    set_target_properties(huxerui_preview PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED TRUE)
    target_compile_options(huxerui_preview PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
endif ()
if (WIN32)
    target_link_libraries(huxerui_preview PRIVATE d3d11 windowsapp windowscodecs ole32 uuid)
elseif (APPLE)
    target_link_libraries(huxerui_preview PRIVATE "-framework AppKit")
elseif (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PREVIEW_GTK REQUIRED IMPORTED_TARGET gtk4)
    target_link_libraries(huxerui_preview PRIVATE PkgConfig::PREVIEW_GTK)
endif ()
file(GENERATE OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/preview-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:huxerui_preview>")
